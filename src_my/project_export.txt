import json
import uuid

from domain.model.game import Game
from domain.model.game_board import GameBoard
from domain.model.game_state import GameState
from datasource.model.game_ds import GameDS
from datasource.model.user_ds import UserDS
from domain.model.user import User


class GameMapper:

    @staticmethod
    def domain_to_ds(game: Game) -> GameDS:

        return GameDS(
            id            = str(game.id),
            state         = game.state.value,
            player_x_id   = str(game.player_x_id) if game.player_x_id else None,
            player_o_id   = str(game.player_o_id) if game.player_o_id else None,
            current_turn  = str(game.current_turn) if game.current_turn else None,
            opponent_type = game.opponent_type,
            board_json    = json.dumps(game.board.matrix),
        )

    @staticmethod
    def ds_to_domain(ds: GameDS) -> Game:

        board = GameBoard(matrix=json.loads(ds.board_json))
        game  = Game(board=board)

        game.id            = uuid.UUID(ds.id)
        game.state         = GameState(ds.state)
        game.player_x_id   = uuid.UUID(ds.player_x_id) if ds.player_x_id else None
        game.player_o_id   = uuid.UUID(ds.player_o_id) if ds.player_o_id else None
        game.current_turn  = uuid.UUID(ds.current_turn) if ds.current_turn else None
        game.opponent_type = ds.opponent_type

        return game


class UserMapper:

    @staticmethod
    def domain_to_ds(user: User) -> UserDS:

        return UserDS(
            id            = str(user.id),
            login         = user.login,
            password_hash = user.password_hash,
        )

    @staticmethod
    def ds_to_domain(ds: UserDS) -> User:

        user               = User(login=ds.login, password='')
        user.id            = uuid.UUID(ds.id)
        user.password_hash = ds.password_hash

        return user

from typing import Optional

from datasource.database import db
from datasource.model.game_ds import GameDS


class GameRepository:

    def save(self, game: GameDS) -> None:
        db.session.merge(game)
        db.session.commit()

    def get_by_id(self, game_id: str) -> Optional[GameDS]:
        return db.session.get(GameDS, game_id)

    def get_all_waiting(self) -> list[GameDS]:
        one_hour_ago = datetime.utcnow() - timedelta(hours=1)

        return GameDS.query.filter(
            GameDS.state      == 'waiting',
            GameDS.created_at >= one_hour_ago,
        ).all()

import uuid
from typing import Optional

from domain.service.impl import GameServiceImpl, check_winner
from domain.model.game import Game
from domain.model.game_board import GameBoard
from domain.model.game_state import GameState
from domain.model.user import User
from datasource.repository.game_repo import GameRepository
from datasource.repository.user_repo import UserRepository
from datasource.mapper.mapper import GameMapper, UserMapper


class GameServiceImplWithRepo(GameServiceImpl):

    def __init__(
        self,
        repository: GameRepository,
        mapper:     GameMapper,
        user_repo:  UserRepository,
    ):
        super().__init__()
        self._repo      = repository
        self._mapper    = mapper
        self._user_repo = user_repo

    def create_game(
        self,
        player_id:     uuid.UUID,
        opponent_type: str = 'computer',
    ) -> Game:
        game               = Game()
        game.player_x_id   = player_id
        game.opponent_type = opponent_type

        if opponent_type == 'computer':
            game.state        = GameState.TURN
            game.current_turn = player_id
        else:
            game.state = GameState.WAITING

        self._repo.save(self._mapper.domain_to_ds(game))

        return game

    def get_available_games(self) -> list[Game]:
        return [
            self._mapper.ds_to_domain(ds)
            for ds in self._repo.get_all_waiting()
        ]

    def join_game(
        self,
        game_id:   str,
        player_id: uuid.UUID,
    ) -> Optional[Game]:
        game_ds = self._repo.get_by_id(game_id)

        if game_ds is None:
            return None
        else:
            game = self._mapper.ds_to_domain(game_ds)

            if game.state != GameState.WAITING:
                return None
            elif game.player_x_id == player_id:
                return None
            else:
                game.player_o_id  = player_id
                game.state        = GameState.TURN
                game.current_turn = game.player_x_id

                self._repo.save(self._mapper.domain_to_ds(game))

                return game

    def get_game(self, game_id: str) -> Optional[Game]:
        game_ds = self._repo.get_by_id(game_id)

        if game_ds is None:
            return None
        else:
            return self._mapper.ds_to_domain(game_ds)

    def get_user(self, user_id: str) -> Optional[User]:
        user_ds = self._user_repo.get_by_id(user_id)

        if user_ds is None:
            return None
        else:
            return UserMapper().ds_to_domain(user_ds)

    def make_move(
        self,
        game_id:   str,
        player_id: uuid.UUID,
        row:       int,
        col:       int,
    ) -> Optional[Game]:
        game_ds = self._repo.get_by_id(game_id)

        if game_ds is None:
            return None
        else:
            game = self._mapper.ds_to_domain(game_ds)

            if game.state != GameState.TURN:
                return None
            elif game.current_turn != player_id:
                return None
            elif game.board.matrix[row][col] != 0:
                return None
            else:
                symbol                      = 1 if game.player_x_id == player_id else 2
                game.board.matrix[row][col] = symbol
                game                        = self._update_state(game, player_id)

                if game.state == GameState.TURN and game.opponent_type == 'computer':
                    game = self._computer_move(game)

                self._repo.save(self._mapper.domain_to_ds(game))

                return game

    def _update_state(self, game: Game, last_player_id: uuid.UUID) -> Game:
        if self.is_game_over(game):
            winner_symbol = check_winner(game.board.matrix)

            if winner_symbol == 0:
                game.state = GameState.DRAW
            else:
                game.state        = GameState.WIN
                game.current_turn = (
                    game.player_x_id if winner_symbol == 1
                    else game.player_o_id
                )
        else:
            game.state        = GameState.TURN
            game.current_turn = (
                game.player_o_id if last_player_id == game.player_x_id
                else game.player_x_id
            )

        return game

    def _computer_move(self, game: Game) -> Game:
        r, c = self.get_next_move(game)

        if r != -1 and game.board.matrix[r][c] == 0:
            game.board.matrix[r][c] = 2

        game = self._update_state(game, game.player_o_id or uuid.uuid4())

        return game

    def save_game_state(self, game: Game) -> None:
        self._repo.save(self._mapper.domain_to_ds(game))

    def get_previous_board(self, game_id: str) -> Optional[GameBoard]:
        game_ds = self._repo.get_by_id(str(game_id))

        if game_ds is None:
            return None
        else:
            return self._mapper.ds_to_domain(game_ds).board

from typing import List

class GameBoard:
    def __init__(self, matrix: List[List[int]] | None = None, size: int = 3):
        if matrix is None:
            self.matrix = [[0] * size for _ in range(size)]
        else:
            self.matrix = [row[:] for row in matrix]
        self.size = len(self.matrix)

    def copy(self) -> "GameBoard":
        return GameBoard(self.matrix)

import base64
import hashlib
import uuid
from typing import Optional

from domain.model.signup_request import SignUpRequest
from domain.service.user_service import UserService
from datasource.model.user_ds import UserDS


class AuthService:

    def __init__(self, user_service: UserService):
        self._user_service = user_service

    @staticmethod
    def _hash_password(password: str) -> str:
        return hashlib.sha256(password.encode()).hexdigest()

    def register(self, request: SignUpRequest) -> bool:
        existing = self._user_service.get_by_login(request.login)

        if existing is not None:
            return False
        else:
            user_ds               = UserDS()
            user_ds.id            = str(uuid.uuid4())
            user_ds.login         = request.login
            user_ds.password_hash = self._hash_password(request.password)

            self._user_service.save(user_ds)

            return True

    def login(self, auth_header: str) -> Optional[uuid.UUID]:
        try:
            encoded         = auth_header.split(' ')[1]
            decoded         = base64.b64decode(encoded).decode('utf-8')
            login, password = decoded.split(':', maxsplit=1)
        except Exception:
            return None

        user_ds = self._user_service.get_by_login(login)

        if user_ds is None:
            return None
        elif user_ds.password_hash != self._hash_password(password):
            return None
        else:
            return uuid.UUID(user_ds.id)

from typing import Tuple

from domain.service.interface import GameServiceInterface
from domain.model.game import Game
from domain.model.game_board import GameBoard


class GameServiceImpl(GameServiceInterface):

    def get_next_move(self, game: Game) -> Tuple[int, int]:
        matrix     = [row[:] for row in game.board.matrix]
        best_score = float('-inf')
        best_move  = (-1, -1)

        for r in range(game.board.size):
            for c in range(game.board.size):
                if matrix[r][c] == 0:
                    matrix[r][c] = 2
                    score        = self._minimax(matrix, depth=0, is_maximizing=False)
                    matrix[r][c] = 0

                    if score > best_score:
                        best_score = score
                        best_move  = (r, c)

        return best_move

    def validate_board(self, game: Game, previous_board: GameBoard) -> bool:
        if game.board.size != previous_board.size:
            return False

        diff_count = 0

        for r in range(game.board.size):
            for c in range(game.board.size):
                if game.board.matrix[r][c] != previous_board.matrix[r][c]:
                    diff_count += 1

        return diff_count == 1

    def is_game_over(self, game: Game) -> bool:
        if check_winner(game.board.matrix) != 0:
            return True
        elif all(cell != 0 for row in game.board.matrix for cell in row):
            return True
        else:
            return False

    def _minimax(self, matrix: list, depth: int, is_maximizing: bool) -> int:
        winner = check_winner(matrix)

        if winner == 2:
            return 10 - depth
        elif winner == 1:
            return depth - 10
        elif all(cell != 0 for row in matrix for cell in row):
            return 0
        elif is_maximizing:
            best = float('-inf')

            for r in range(3):
                for c in range(3):
                    if matrix[r][c] == 0:
                        matrix[r][c] = 2
                        score        = self._minimax(matrix, depth + 1, False)
                        matrix[r][c] = 0
                        best         = max(best, score)

            return best
        else:
            best = float('inf')

            for r in range(3):
                for c in range(3):
                    if matrix[r][c] == 0:
                        matrix[r][c] = 1
                        score        = self._minimax(matrix, depth + 1, True)
                        matrix[r][c] = 0
                        best         = min(best, score)

            return best


def check_winner(matrix: list) -> int:
    lines = [
        [(0,0), (0,1), (0,2)],
        [(1,0), (1,1), (1,2)],
        [(2,0), (2,1), (2,2)],
        [(0,0), (1,0), (2,0)],
        [(0,1), (1,1), (2,1)],
        [(0,2), (1,2), (2,2)],
        [(0,0), (1,1), (2,2)],
        [(0,2), (1,1), (2,0)],
    ]

    for line in lines:
        cells = [matrix[r][c] for r, c in line]

        if cells[0] != 0 and cells[0] == cells[1] == cells[2]:
            return cells[0]

    return 0

import uuid
from typing import Optional, Callable
from functools import wraps

from flask import request, jsonify

from domain.service.auth_service import AuthService


class UserAuthenticator:

    def __init__(self, auth_service: AuthService):
        self._auth_service = auth_service

    def get_current_user_id(self) -> Optional[uuid.UUID]:
        auth_header: Optional[str] = request.headers.get('Authorization')

        if auth_header is None:
            return None

        return self._auth_service.login(auth_header)

    def required(self, f: Callable) -> Callable:
        @wraps(f)
        def decorated(*args, **kwargs):
            user_id = self.get_current_user_id()

            if user_id is None:
                response             = jsonify({'error': 'Unauthorized'})
                response.status_code = 401
                response.headers['WWW-Authenticate'] = 'Basic realm="Login required"'
                return response
            else:
                return f(*args, user_id=user_id, **kwargs)

        return decorated

from flask import Blueprint, request, jsonify

from domain.service.auth_service import AuthService
from domain.model.signup_request import SignUpRequest


def create_auth_controller(auth_service: AuthService) -> Blueprint:

    bp = Blueprint('auth', __name__, url_prefix='/auth')

    @bp.route('/register', methods=['POST'])
    def register():
        data = request.get_json()

        if not data or 'login' not in data or 'password' not in data:
            return jsonify({'error': 'login and password required'}), 400

        signup = SignUpRequest(
            login    = data['login'],
            password = data['password'],
        )

        success = auth_service.register(signup)

        if not success:
            return jsonify({'error': 'login already taken'}), 409

        return jsonify({'message': 'registered successfully'}), 201

    @bp.route('/login', methods=['POST'])
    def login():
        auth_header = request.headers.get('Authorization')

        if not auth_header:
            return jsonify({'error': 'Authorization header required'}), 401

        user_id = auth_service.login(auth_header)

        if user_id is None:
            return jsonify({'error': 'invalid login or password'}), 401

        return jsonify({'user_id': str(user_id)}), 200

    return bp

import os
import uuid
from flask import Blueprint, request, jsonify, send_from_directory

from domain.service.interface import GameServiceInterface
from web.auth.authenticator import UserAuthenticator
from web.mapper.mapper import GameWebMapper


def create_game_controller(
    service:       GameServiceInterface,
    authenticator: UserAuthenticator,
) -> tuple:

    game_bp = Blueprint('game', __name__, url_prefix='/games')
    user_bp = Blueprint('user', __name__, url_prefix='/users')

    static_dir = os.path.join(os.path.dirname(__file__), '..', 'static')

    @game_bp.route('/', defaults={'path': ''})
    @game_bp.route('/<path:path>')
    def serve_frontend(path):
        if path == '' or path == '/':
            return send_from_directory(static_dir, 'index.html')
        else:
            return send_from_directory(static_dir, path)

    @game_bp.route('', methods=['POST'])
    @authenticator.required
    def create_game(user_id: uuid.UUID):
        data          = request.get_json() or {}
        opponent_type = data.get('opponent_type', 'computer')

        if opponent_type not in ('computer', 'user'):
            return jsonify({'error': 'opponent_type must be computer or user'}), 400
        else:
            game = service.create_game(
                player_id     = user_id,
                opponent_type = opponent_type,
            )
            return jsonify(GameWebMapper.domain_to_web(game).to_dict()), 201

    @game_bp.route('', methods=['GET'])
    @authenticator.required
    def get_games(user_id: uuid.UUID):
        games = service.get_available_games()
        return jsonify([
            GameWebMapper.domain_to_web(g).to_dict() for g in games
        ]), 200

    @game_bp.route('/<string:game_id>/join', methods=['POST'])
    @authenticator.required
    def join_game(game_id: str, user_id: uuid.UUID):
        game = service.join_game(
            game_id   = game_id,
            player_id = user_id,
        )

        if game is None:
            return jsonify({'error': 'game not found or already started'}), 404
        else:
            return jsonify(GameWebMapper.domain_to_web(game).to_dict()), 200

    @game_bp.route('/<string:game_id>', methods=['GET'])
    @authenticator.required
    def get_game(game_id: str, user_id: uuid.UUID):
        game = service.get_game(game_id)

        if game is None:
            return jsonify({'error': 'game not found'}), 404
        else:
            return jsonify(GameWebMapper.domain_to_web(game).to_dict()), 200

    @game_bp.route('/<string:game_id>', methods=['PATCH'])
    @authenticator.required
    def make_move(game_id: str, user_id: uuid.UUID):
        data = request.get_json()

        if not data or 'row' not in data or 'col' not in data:
            return jsonify({'error': 'row and col required'}), 400
        else:
            game = service.make_move(
                game_id   = game_id,
                player_id = user_id,
                row       = data['row'],
                col       = data['col'],
            )

            if game is None:
                return jsonify({'error': 'game not found or move not allowed'}), 400
            else:
                return jsonify(GameWebMapper.domain_to_web(game).to_dict()), 200

    @user_bp.route('/<string:user_uuid>', methods=['GET'])
    @authenticator.required
    def get_user(user_uuid: str, user_id: uuid.UUID):
        user = service.get_user(user_uuid)

        if user is None:
            return jsonify({'error': 'user not found'}), 404
        else:
            return jsonify({
                'id':    str(user.id),
                'login': user.login,
            }), 200

    return game_bp, user_bp

const state = {
  login:     null,   
  password:  null,   
  userId:    null,   
  gameId:    null,   
  game:      null,   
  mySymbol:  null,   
  pollTimer: null,   
  lastState: null,   
  lastBoard: null,  
};

const API = window.location.origin;
const POLL_INTERVAL = 2000; 





function authHeader() {
  if (!state.login) return {};
  return { 'Authorization': 'Basic ' + btoa(state.login + ':' + state.password) };
}


async function api(method, path, body = null, silent = false) {
  const opts = {
    method,
    headers: { 'Content-Type': 'application/json', ...authHeader() },
  };
  if (body) opts.body = JSON.stringify(body);

  if (!silent) log('→ ' + method + ' ' + path, 'dim');

  try {
    const resp = await fetch(API + path, opts);
    const data = await resp.json().catch(() => ({}));
    if (!silent) log('← ' + resp.status + ' ' + JSON.stringify(data).slice(0, 120), resp.ok ? 'ok' : 'err');
    return { ok: resp.ok, status: resp.status, data };
  } catch (e) {
    if (!silent) log('✕ ОШИБКА: ' + e.message, 'err');
    return { ok: false, data: {} };
  }
}



function startPolling() {
  stopPolling();
  document.getElementById('poll-indicator').classList.remove('hidden');
  state.pollTimer = setInterval(pollGame, POLL_INTERVAL);
}

function stopPolling() {
  if (state.pollTimer) {
    clearInterval(state.pollTimer);
    state.pollTimer = null;
  }
  document.getElementById('poll-indicator').classList.add('hidden');
}

async function pollGame() {
  if (!state.gameId || !state.login) return;


  const dot = document.getElementById('poll-dot');
  dot.classList.add('active');
  setTimeout(() => dot.classList.remove('active'), 300);


  const r = await api('GET', '/games/' + state.gameId, null, true);
  if (!r.ok) return;

  const g        = r.data;
  const boardStr = JSON.stringify(g.board.matrix);


  if (g.state !== state.lastState || boardStr !== state.lastBoard) {
    state.game      = g;
    state.lastBoard = boardStr;
    renderGame();


    if (g.state !== state.lastState) {
      if (state.lastState === 'waiting' && g.state === 'turn') {
        log('✓ Второй игрок присоединился! Игра началась.', 'ok');
      } else if (g.state === 'turn' && g.current_turn === state.userId) {
        log('▶ Твой ход!', 'ok');
      } else if (g.state === 'turn' && g.current_turn !== state.userId) {
        log('⏳ Ход соперника...', 'info');
      } else if (g.state === 'win') {
        const iWon = g.current_turn === state.userId;
        log(iWon ? '🏆 Ты победил!' : '✕ Ты проиграл', iWon ? 'ok' : 'err');
        stopPolling();
        return;
      } else if (g.state === 'draw') {
        log('≈ Ничья!', 'info');
        stopPolling();
        return;
      }
    }

    state.lastState = g.state;
  }
}




async function doLogin() {
  const login    = document.getElementById('login-input').value.trim();
  const password = document.getElementById('password-input').value;
  if (!login || !password) return log('✕ Введи логин и пароль', 'err');

  state.login    = login;
  state.password = password;

  const r = await api('POST', '/auth/login');

  if (r.ok) {
    state.userId = r.data.user_id;
    document.getElementById('auth-status').className   = 'auth-status logged-in';
    document.getElementById('auth-status').textContent = '● ВОШЁЛ: ' + login;
    document.getElementById('user-uuid-box').classList.remove('hidden');
    document.getElementById('user-uuid-val').textContent = state.userId;
    log('✓ Авторизован как ' + login, 'ok');
  } else {
    state.login = state.password = null;
    log('✕ Неверный логин или пароль', 'err');
  }
}

function doLogout() {
  stopPolling();
  state.login = state.password = state.userId = null;
  document.getElementById('auth-status').className   = 'auth-status logged-out';
  document.getElementById('auth-status').textContent = '○ НЕ АВТОРИЗОВАН';
  document.getElementById('user-uuid-box').classList.add('hidden');
  log('✓ Вышел из аккаунта', 'info');
}

async function doRegister() {
  const login    = document.getElementById('reg-login').value.trim();
  const password = document.getElementById('reg-password').value;
  if (!login || !password) return log('✕ Введи логин и пароль', 'err');

  const r = await api('POST', '/auth/register', { login, password });
  if (r.ok) {
    log('✓ Пользователь ' + login + ' зарегистрирован', 'ok');
  } else {
    log('✕ ' + (r.data.error || 'Ошибка регистрации'), 'err');
  }
}



async function createGame() {
  if (!state.login) return log('✕ Войди в аккаунт', 'err');

  const opponent_type = document.getElementById('opponent-type').value;
  const r = await api('POST', '/games', { opponent_type });

  if (r.ok) {
    state.gameId    = r.data.id;
    state.game      = r.data;
    state.lastState = r.data.state;
    state.lastBoard = JSON.stringify(r.data.board.matrix);
    state.mySymbol  = 1; 

    renderGame();
    startPolling();
    log('✓ Игра создана — ты играешь за X', 'ok');

    if (opponent_type === 'user') {
      log('⏳ Ожидаем второго игрока...', 'info');
    }
  }
}

async function joinGame(gameId) {
  if (!state.login) return log('✕ Войди в аккаунт', 'err');

  const r = await api('POST', '/games/' + gameId + '/join');

  if (r.ok) {
    state.gameId    = gameId;
    state.game      = r.data;
    state.lastState = r.data.state;
    state.lastBoard = JSON.stringify(r.data.board.matrix);
    state.mySymbol  = 2; 

    renderGame();
    startPolling();
    log('✓ Присоединился к игре — ты играешь за O', 'ok');
  }
}

async function makeMove() {
  if (!state.gameId) return log('✕ Нет активной игры', 'err');

  const row = parseInt(document.getElementById('move-row').value);
  const col = parseInt(document.getElementById('move-col').value);
  const r   = await api('PATCH', '/games/' + state.gameId, { row, col });

  if (r.ok) {
    state.game      = r.data;
    state.lastState = r.data.state;
    state.lastBoard = JSON.stringify(r.data.board.matrix);
    renderGame();
  }
}

async function makeCellMove(row, col) {
  document.getElementById('move-row').value = row;
  document.getElementById('move-col').value = col;
  await makeMove();
}

async function refreshGame() {
  if (!state.gameId) return log('✕ Нет активной игры', 'err');
  const r = await api('GET', '/games/' + state.gameId);
  if (r.ok) { state.game = r.data; renderGame(); }
}

function clearGame() {
  stopPolling();
  state.gameId    = null;
  state.game      = null;
  state.mySymbol  = null;
  state.lastState = null;
  state.lastBoard = null;

  document.getElementById('game-id-box').classList.add('hidden');
  document.getElementById('my-symbol').classList.add('hidden');

  renderGame();
  log('✓ Игра сброшена', 'info');
}

async function loadGames() {
  if (!state.login) return log('✕ Войди в аккаунт', 'err');

  const r    = await api('GET', '/games');
  const list = document.getElementById('games-list');

  if (!r.ok) return;

  if (!r.data.length) {
    list.innerHTML = '<div class="empty-msg">Нет доступных игр</div>';
    return;
  }

  list.innerHTML = r.data.map(g => `
    <div class="game-item">
      <div>

        <span class="badge badge-waiting">WAITING</span>

      </div>

    </div>
  `).join('');
}

function renderGame() {
  const board    = document.getElementById('board');
  const statusEl = document.getElementById('game-status');
  const idBox    = document.getElementById('game-id-box');
  const symBox   = document.getElementById('my-symbol');

  if (!state.game) {
    board.innerHTML = '';
    for (let i = 0; i < 9; i++) {
      const cell     = document.createElement('div');
      cell.className = 'cell blocked';
      board.appendChild(cell);
    }
    statusEl.className   = 'game-status gs-idle';
    statusEl.textContent = 'ИГРА НЕ НАЧАТА';
    return;
  }

  const g        = state.game;
  const matrix   = g.board.matrix;
  const gameOver = g.state === 'win' || g.state === 'draw';
  const isMyTurn = g.state === 'turn' && g.current_turn === state.userId;

  idBox.classList.remove('hidden');
  document.getElementById('game-uuid-val').textContent = g.id;

  if (state.mySymbol) {
    symBox.classList.remove('hidden');
    symBox.className   = 'my-symbol-box ' + (state.mySymbol === 1 ? 'x' : 'o');
    symBox.textContent = 'ТЫ ИГРАЕШЬ ЗА: ' + (state.mySymbol === 1 ? 'X ✕' : 'O ○');
  }

  statusEl.className = 'game-status ';
  if (g.state === 'waiting') {
    statusEl.className  += 'gs-waiting';
    statusEl.textContent = '⏳ ОЖИДАНИЕ ВТОРОГО ИГРОКА...';
  } else if (g.state === 'turn' && isMyTurn) {
    statusEl.className  += 'gs-turn';
    statusEl.textContent = '▶ ТВОЙ ХОД ' + (state.mySymbol === 1 ? '(X)' : '(O)');
  } else if (g.state === 'turn' && !isMyTurn) {
    statusEl.className  += 'gs-enemy';
    statusEl.textContent = '⏳ ХОД СОПЕРНИКА...';
  } else if (g.state === 'win') {
    const iWon          = g.current_turn === state.userId;
    statusEl.className  += iWon ? 'gs-win' : 'gs-lose';
    statusEl.textContent = iWon ? '🏆 ТЫ ПОБЕДИЛ!' : '✕ ТЫ ПРОИГРАЛ';
  } else if (g.state === 'draw') {
    statusEl.className  += 'gs-draw';
    statusEl.textContent = '≈ НИЧЬЯ';
  }

  board.innerHTML = '';
  for (let r = 0; r < 3; r++) {
    for (let c = 0; c < 3; c++) {
      const val      = matrix[r][c];
      const sym      = val === 1 ? 'X' : val === 2 ? 'O' : '';
      const symClass = val === 1 ? ' x' : val === 2 ? ' o' : '';

      const canClick = val === 0 && isMyTurn && !gameOver;

      const cell       = document.createElement('div');
      cell.className   = 'cell' + symClass + (canClick ? ' clickable' : ' blocked');
      cell.textContent = sym;

      if (canClick) {
        cell.onclick = ((row, col) => () => makeCellMove(row, col))(r, c);
      }

      board.appendChild(cell);
    }
  }
}
