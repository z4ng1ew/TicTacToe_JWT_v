import json
from typing import List, Optional
from sqlalchemy import Integer, String
from sqlalchemy.orm import Mapped, mapped_column
from datasource.db import Base


class GameSessionDB(Base):
    __tablename__: str = "game_sessions"
    session_id: Mapped[str] = mapped_column(String(36), primary_key=True)
    side_length: Mapped[int] = mapped_column(Integer, default=3)
    grid_json: Mapped[str] = mapped_column(String)
    state: Mapped[str] = mapped_column(String, default="waiting")
    player_x_id: Mapped[Optional[str]] = mapped_column(String(36), nullable=True)
    player_o_id: Mapped[Optional[str]] = mapped_column(String(36), nullable=True)
    mode: Mapped[str] = mapped_column(String, default="pve")

    def to_grid(self) -> List[List[int]]:
        return json.loads(self.grid_json)

import json
from typing import Any, Dict, List, Optional
from sqlalchemy.orm import Session
from datasource.db import get_session
from datasource.model.game_db import GameSessionDB


class GameRepository:
    def save_match(
        self,
        session_id: str,
        grid: List[List[int]],
        side_length: int,
        state: str = "waiting",
        player_x_id: Optional[str] = None,
        player_o_id: Optional[str] = None,
        mode: str = "pve",
    ) -> None:
        db_session: Session = get_session()
        try:
            existing: Optional[GameSessionDB] = (
                db_session.query(GameSessionDB).filter_by(session_id=session_id).first()
            )
            if existing is not None:
                existing.side_length = side_length
                existing.grid_json = json.dumps(grid)
                existing.state = state
                existing.player_x_id = player_x_id
                existing.player_o_id = player_o_id
                existing.mode = mode
            else:
                new_record: GameSessionDB = GameSessionDB(
                    session_id=session_id,
                    side_length=side_length,
                    grid_json=json.dumps(grid),
                    state=state,
                    player_x_id=player_x_id,
                    player_o_id=player_o_id,
                    mode=mode,
                )
                db_session.add(new_record)
            db_session.commit()
        except Exception:
            db_session.rollback()
            raise
        finally:
            db_session.close()

    def load_match(self, session_id: str) -> Optional[Dict[str, Any]]:
        db_session: Session = get_session()
        try:
            record: Optional[GameSessionDB] = (
                db_session.query(GameSessionDB).filter_by(session_id=session_id).first()
            )
            if record is not None:
                return {
                    "session_id": record.session_id,
                    "side_length": record.side_length,
                    "grid": record.to_grid(),
                    "state": record.state,
                    "player_x_id": record.player_x_id,
                    "player_o_id": record.player_o_id,
                    "mode": record.mode,
                }
            else:
                return None
        finally:
            db_session.close()

    def find_waiting_games(self) -> List[Dict[str, Any]]:
        db_session: Session = get_session()
        try:
            records: List[GameSessionDB] = (
                db_session.query(GameSessionDB).filter_by(state="waiting", mode="pvp").all()
            )
            return [
                {
                    "id": r.session_id,
                    "session_id": r.session_id,
                    "player_x_id": r.player_x_id,
                }
                for r in records
            ]
        finally:
            db_session.close()

    def join_game(self, session_id: str, player_id: str) -> bool:
        db_session: Session = get_session()
        try:
            record: Optional[GameSessionDB] = (
                db_session.query(GameSessionDB).filter_by(session_id=session_id).first()
            )
            if record is None:
                return False
            elif record.state != "waiting":
                return False
            elif record.mode != "pvp":
                return False
            elif record.player_x_id is None:
                record.player_x_id = player_id
                record.state = "player_x_turn"
                db_session.commit()
                return True
            elif record.player_o_id is None and record.player_x_id != player_id:
                record.player_o_id = player_id
                record.state = "player_x_turn"
                db_session.commit()
                return True
            else:
                return False
        except Exception:
            db_session.rollback()
            raise
        finally:
            db_session.close()

class DependencyBox:
    def __init__(self) -> None:
        self._singletons: Dict[str, Any] = {}
        self._factories: Dict[str, Callable[[], Any]] = {}
        self._setup()

    def _setup(self) -> None:
        self.register_singleton("game_repo", lambda: GameRepository())
        self.register_singleton("user_repo", lambda: UserRepository())
        self.register_singleton("logic", lambda: LogicWithArchive(self.get("game_repo")))
        self.register_singleton("user_svc", lambda: UserService(self.get("user_repo")))
        self.register_singleton("auth_svc", lambda: AuthService(self.get("user_svc")))
        self.register_singleton("web_app", self._build_web_app)

    def _build_web_app(self) -> Flask:
        app: Flask = build_app(
            logic_service=self.get("logic"),
            auth_service=self.get("auth_svc"),
            game_repo=self.get("game_repo"),
            user_svc=self.get("user_svc"),
        )
        app.register_blueprint(create_auth_routes())
        return app

    def register_singleton(self, name: str, factory: Callable[[], Any]) -> None:
        self._factories[name] = factory
        self._singletons[name] = factory()

    def get(self, name: str) -> Any:
        if name in self._singletons:
            return self._singletons[name]
        elif name in self._factories:
            return self._factories[name]()
        else:
            raise ValueError(f"Компонент '{name}' не найден в реестре")

import uuid
from enum import Enum
from typing import Optional
from domain.model.board import GameField


class GameState(Enum):
    WAITING = "waiting"
    PLAYER_X_TURN = "player_x_turn"
    PLAYER_O_TURN = "player_o_turn"
    DRAW = "draw"
    WIN_X = "win_x"
    WIN_O = "win_o"


class MatchSession:
    def __init__(
        self,
        current_field: Optional[GameField] = None,
        player_x_id: Optional[str] = None,
        player_o_id: Optional[str] = None,
        state: GameState = GameState.WAITING,
        mode: str = "pve",
        session_id: Optional[uuid.UUID] = None,
    ) -> None:
        self.session_id: uuid.UUID = session_id if session_id is not None else uuid.uuid4()
        self.field: GameField = current_field if current_field is not None else GameField()
        self.player_x_id: Optional[str] = player_x_id
        self.player_o_id: Optional[str] = player_o_id
        self.state: GameState = state
        self.mode: str = mode

    def get_current_player_id(self) -> Optional[str]:
        if self.state == GameState.PLAYER_X_TURN:
            return self.player_x_id
        elif self.state == GameState.PLAYER_O_TURN:
            return self.player_o_id
        else:
            return None

    def get_symbol_for_player(self, player_id: str) -> Optional[int]:
        if player_id == self.player_x_id:
            return 1
        elif player_id == self.player_o_id:
            return 2
        else:
            return None

from typing import List, Tuple
from domain.model.board import GameField
from domain.model.game import GameState, MatchSession


class AiEngine:
    def calculate_ai_turn(self, session: MatchSession) -> Tuple[int, int]:
        grid: List[List[int]] = [row.copy() for row in session.field.grid]
        top_score: float = float("-inf")
        best_cell: Tuple[int, int] = (-1, -1)

        for r in range(session.field.side_length):
            for c in range(session.field.side_length):
                if grid[r][c] == 0:
                    grid[r][c] = 2
                    score: int = self._run_minimax(grid, depth=0, is_max=False)
                    grid[r][c] = 0
                    if score > top_score:
                        top_score = score
                        best_cell = (r, c)
                    else:
                        continue
                else:
                    continue
        return best_cell

    def check_board_changes(self, current: MatchSession, previous: GameField) -> bool:
        if current.field.side_length != previous.side_length:
            return False
        else:
            change_count: int = 0
            for r in range(current.field.side_length):
                for c in range(current.field.side_length):
                    if current.field.grid[r][c] != previous.grid[r][c]:
                        change_count += 1
                    else:
                        continue
            if change_count == 1:
                return True
            else:
                return False

    def verify_end_state(self, session: MatchSession) -> bool:
        winner_code: int = self._find_winner(session.field.grid)
        if winner_code != 0:
            return True
        elif all(cell != 0 for row in session.field.grid for cell in row):
            return True
        else:
            return False

    def update_game_state(self, session: MatchSession) -> None:
        winner: int = self._find_winner(session.field.grid)
        if winner == 1:
            session.state = GameState.WIN_X
        elif winner == 2:
            session.state = GameState.WIN_O
        elif all(cell != 0 for row in session.field.grid for cell in row):
            session.state = GameState.DRAW
        elif session.state == GameState.PLAYER_X_TURN:
            session.state = GameState.PLAYER_O_TURN
        elif session.state == GameState.PLAYER_O_TURN:
            session.state = GameState.PLAYER_X_TURN
        else:
            session.state = GameState.PLAYER_X_TURN

    def _run_minimax(self, grid: List[List[int]], depth: int, is_max: bool) -> int:
        winner_code: int = self._find_winner(grid)
        if winner_code == 2:
            return 10 - depth
        elif winner_code == 1:
            return depth - 10
        elif all(cell != 0 for row in grid for cell in row):
            return 0
        elif is_max:
            best_max: int = -1000
            for r in range(3):
                for c in range(3):
                    if grid[r][c] == 0:
                        grid[r][c] = 2
                        score: int = self._run_minimax(grid, depth + 1, False)
                        grid[r][c] = 0
                        if score > best_max:
                            best_max = score
                        else:
                            continue
                    else:
                        continue
            return best_max
        else:
            best_min: int = 1000
            for r in range(3):
                for c in range(3):
                    if grid[r][c] == 0:
                        grid[r][c] = 1
                        score = self._run_minimax(grid, depth + 1, True)
                        grid[r][c] = 0
                        if score < best_min:
                            best_min = score
                        else:
                            continue
                    else:
                        continue
            return best_min

    @staticmethod
    def _find_winner(grid: List[List[int]]) -> int:
        lines: List[List[Tuple[int, int]]] = [
            [(0, 0), (0, 1), (0, 2)],
            [(1, 0), (1, 1), (1, 2)],
            [(2, 0), (2, 1), (2, 2)],
            [(0, 0), (1, 0), (2, 0)],
            [(0, 1), (1, 1), (2, 1)],
            [(0, 2), (1, 2), (2, 2)],
            [(0, 0), (1, 1), (2, 2)],
            [(0, 2), (1, 1), (2, 0)],
        ]
        for line in lines:
            cells: List[int] = [grid[r][c] for r, c in line]
            if cells[0] != 0 and cells[0] == cells[1] == cells[2]:
                return cells[0]
            else:
                continue
        return 0

import base64
from typing import Optional
from domain.service.user_service import UserService


class AuthService:
    def __init__(self, user_service: UserService) -> None:
        self._user_service: UserService = user_service

    def register(self, login: str, password: str) -> bool:
        if not login or not password:
            return False
        else:
            return self._user_service.create_user(login, password)

    def authorize(self, auth_header: Optional[str]) -> Optional[str]:
        if not auth_header:
            return None
        elif not auth_header.startswith("Basic "):
            return None
        else:
            try:
                encoded: str = auth_header.split(" ", 1)[1]
                decoded_str: str = base64.b64decode(encoded).decode("utf-8")
                parts: list[str] = decoded_str.split(":", 1)
                if len(parts) != 2:
                    return None
                else:
                    login: str = parts[0]
                    password: str = parts[1]
                    return self._user_service.verify_credentials(login, password)
            except Exception:
                return None

import uuid
from typing import Optional
from werkzeug.security import check_password_hash, generate_password_hash
from datasource.model.user_db import UserDB
from datasource.repository.user_repository import UserRepository


class UserService:
    def __init__(self, repository: UserRepository) -> None:
        self._repository: UserRepository = repository

    def create_user(self, login: str, raw_password: str) -> bool:
        existing: Optional[UserDB] = self._repository.find_by_login(login)
        if existing is not None:
            return False
        else:
            new_id: str = str(uuid.uuid4())
            hashed: str = generate_password_hash(raw_password)
            self._repository.save_user(new_id, login, hashed)
            return True

    def verify_credentials(self, login: str, raw_password: str) -> Optional[str]:
        record: Optional[UserDB] = self._repository.find_by_login(login)
        if record is None:
            return None
        elif check_password_hash(record.password_hash, raw_password):
            return record.user_id
        else:
            return None

    def find_by_id(self, user_id: str) -> Optional[UserDB]:
        return self._repository.find_by_id(user_id)

from typing import Any, Dict, Optional, Tuple
from flask import Blueprint, current_app, jsonify, request
from domain.service.auth_service import AuthService
from web.model.auth_request import SignUpRequest


def create_auth_routes() -> Blueprint:
    bp: Blueprint = Blueprint("auth", __name__)

    @bp.route("/signup", methods=["POST"])
    def signup_endpoint() -> Tuple[Any, int]:
        service: AuthService = current_app.config["auth_service"]
        payload: Optional[Dict[str, Any]] = request.get_json(silent=True)
        if payload is None:
            return jsonify({"error": "Пустое или некорректное тело запроса"}), 400
        elif "login" not in payload or "password" not in payload:
            return jsonify({"error": "Отсутствуют поля login/password"}), 400
        else:
            req: SignUpRequest = SignUpRequest(login=payload["login"], password=payload["password"])
            success: bool = service.register(req.login, req.password)
            if success:
                return jsonify({"status": "registered"}), 201
            else:
                return jsonify({"error": "Логин уже занят или данные пусты"}), 409

    @bp.route("/login", methods=["POST"])
    def login_endpoint() -> Tuple[Any, int]:
        service: AuthService = current_app.config["auth_service"]
        header: Optional[str] = request.headers.get("Authorization")
        if not header:
            return jsonify({"error": "Требуется заголовок Authorization: Basic"}), 401
        else:
            user_id: Optional[str] = service.authorize(header)
            if user_id is not None:
                return jsonify({"user_id": user_id}), 200
            else:
                return jsonify({"error": "Неверный логин или пароль"}), 401

    return bp

import uuid
from typing import Any, Dict
from domain.model.board import GameField
from domain.model.game import MatchSession
from web.model.board_view import FieldView
from web.model.game_view import SessionView


class ViewDomainTranslator:
    @staticmethod
    def domain_to_view(session: MatchSession) -> SessionView:
        board_view: FieldView = FieldView(
            grid=[row.copy() for row in session.field.grid],
            side_length=session.field.side_length,
        )
        return SessionView(id=str(session.session_id), board=board_view)

    @staticmethod
    def view_to_domain(view_session: SessionView) -> MatchSession:
        board_data: Dict[str, Any] = (
            {"grid": view_session.board.grid, "size": view_session.board.side_length}
            if isinstance(view_session.board, FieldView)
            else view_session.board
        )
        grid_data = board_data["grid"]
        size_data = board_data.get("size", 3)
        domain_board: GameField = GameField(grid=[row.copy() for row in grid_data], side_length=size_data)
        return MatchSession(
            current_field=domain_board,
            session_id=uuid.UUID(view_session.id),
        )

from functools import wraps
from typing import Any, Callable, Optional
from flask import g, jsonify, request
from domain.service.auth_service import AuthService


class UserAuthenticator:
    def __init__(self, service: AuthService) -> None:
        self._service: AuthService = service

    def authenticate(self, func: Callable[..., Any]) -> Callable[..., Any]:
        @wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            header: Optional[str] = request.headers.get("Authorization")
            if header is None:
                return jsonify({"error": "Требуется авторизация"}), 401
            else:
                user_id: Optional[str] = self._service.authorize(header)
                if user_id is None:
                    return jsonify({"error": "Неверные учётные данные"}), 401
                else:
                    g.user_id = user_id
                    return func(*args, **kwargs)
        return wrapper

    @application.route("/game/<string:game_id>", methods=["POST"])
    @authenticator.authenticate
    def process_turn(game_id: str) -> Tuple[Any, int]:
        payload: Optional[Dict[str, Any]] = request.get_json(silent=True)
        if payload is None or not isinstance(payload, dict):
            abort(400, description="Тело запроса должно быть JSON-объектом")
        else:
            stored: Optional[Dict[str, Any]] = game_repo.load_match(game_id)
            if stored is None:
                abort(404, description="Игра не найдена")
            else:
                board_payload: Dict[str, Any] = payload.get("board", {})
                grid_payload: List[List[int]] = board_payload.get("grid", stored["grid"])

                domain_session: MatchSession = MatchSession(
                    current_field=GameField(grid=grid_payload, side_length=stored["side_length"]),
                    player_x_id=stored["player_x_id"],
                    player_o_id=stored["player_o_id"],
                    state=GameState(stored["state"]),
                    mode=stored["mode"],
                )
                import uuid as _uuid
                domain_session.session_id = _uuid.UUID(stored["session_id"])

                prev_board: Optional[GameField] = logic_service.load_previous_field(domain_session.session_id)
                if prev_board is not None and not logic_service.check_board_changes(domain_session, prev_board):
                    abort(400, description="Проверка поля не пройдена: изменены старые ходы")
                else:
                    pass

                if logic_service.verify_end_state(domain_session):
                    logic_service.update_game_state(domain_session)
                    return _respond(domain_session)
                elif domain_session.mode == "pve":
                    logic_service.update_game_state(domain_session)
                    if not logic_service.verify_end_state(domain_session):
                        row, col = logic_service.calculate_ai_turn(domain_session)
                        if row != -1 and domain_session.field.grid[row][col] == 0:
                            domain_session.field.grid[row][col] = 2
                            logic_service.update_game_state(domain_session)
                        else:
                            abort(500, description="Невозможно сделать ход ИИ")
                    else:
                        pass
                    return _respond(domain_session)
                else:
                    logic_service.update_game_state(domain_session)
                    return _respond(domain_session)

    @application.route("/user/<string:user_id>", methods=["GET"])
    @authenticator.authenticate
    def get_user_info(user_id: str) -> Tuple[Any, int]:
        record = user_svc.find_by_id(user_id)
        if record is None:
            abort(404, description="Пользователь не найден")
        else:
            return jsonify({"user_id": record.user_id, "login": record.login}), 200

    return application

                refreshGameStatus(data.state);
            } catch (e) {
                field[r][c] = 0;
                drawBoard();
                setStatus(`Ошибка: ${e.message}`, '');
            }
        }

        function refreshGameStatus(state) {
            if (state === 'win_x') { setStatus(currentGame.mySymbol === 'X' ? '🎉 Победа!' : '💻 Поражение', currentGame.mySymbol === 'X' ? 'win' : 'lose'); active = false; stopAutoRefresh(); }
            else if (state === 'win_o') { setStatus(currentGame.mySymbol === 'O' ? '🎉 Победа!' : '💻 Поражение', currentGame.mySymbol === 'O' ? 'win' : 'lose'); active = false; stopAutoRefresh(); }
            else if (state === 'draw') { setStatus('🤝 Ничья!', 'draw'); active = false; stopAutoRefresh(); }
            else if (state === 'player_x_turn') { active = currentGame.mySymbol === 'X'; setStatus(active ? 'Твой ход (X)' : 'Ход соперника (X)', 'turn'); }
            else if (state === 'player_o_turn') { active = currentGame.mySymbol === 'O'; setStatus(active ? 'Твой ход (O)' : 'Ход соперника (O)', 'turn'); }
            else if (state === 'waiting') { setStatus('Ожидание соперника...', 'waiting'); active = false; }
            else { setStatus(state, ''); }
            drawBoard();
        }

        async function fetchGameState(gameId) {
            if (!currentGame || currentGame.id !== gameId) return;
            try {
                const data = await api(`/game/${gameId}`, 'GET');
                field = data.board.grid.map(row => [...row]);
                currentGame.state = data.state;
                if (data.player_x_id === currentUser.userId) currentGame.mySymbol = 'X';
                else if (data.player_o_id === currentUser.userId) currentGame.mySymbol = 'O';
                refreshGameStatus(data.state);
            } catch (e) { /* silent */ }
        }

        function startAutoRefresh(gameId) {
            stopAutoRefresh();
            autoRefresh = setInterval(() => fetchGameState(gameId), 2000);
        }
        function stopAutoRefresh() {
            if (autoRefresh) { clearInterval(autoRefresh); autoRefresh = null; }
        }

        async function refreshLobby() {
            if (!currentUser) {
                document.getElementById('lobbyList').innerHTML = '<div class="game-item"><small>🔐 Войдите для доступа</small></div>';
                return;
            }
            try {
                const data = await api('/games/available', 'GET');
                const games = data.games || [];
                const list = document.getElementById('lobbyList');
                if (games.length === 0) {
                    list.innerHTML = '<div class="game-item"><small>📭 Нет доступных игр</small></div>';
                } else {
                    list.innerHTML = games.map(g => `
                        <div class="game-item">
                            <div><strong>ID:</strong> <code>${g.id.slice(0, 8)}...</code>
                            ${g.player_x_id ? `<br><small>Создатель: ${g.player_x_id.slice(0, 8)}...</small>` : ''}
                            </div>
                            <button class="btn secondary" onclick="joinGame('${g.id}')">🎮 Присоединиться</button>
                        </div>
                    `).join('');
                }
            } catch (e) {
                document.getElementById('lobbyList').innerHTML = `<div class="game-item"><small class="err">❌ ${e.message}</small></div>`;
            }
        }

        async function joinGame(gameId) {
            if (!currentUser) { alert('Сначала войдите'); return; }
            try {
                const data = await api(`/game/${gameId}/join`, 'POST', {});
                currentGame = {
                    id: data.id,
                    mode: 'pvp',
                    state: data.state,
                    mySymbol: data.player_x_id === currentUser.userId ? 'X' : 'O'
                };
                field = data.board.grid.map(row => [...row]);
                drawBoard();
                document.getElementById('sid').textContent = `ID: ${data.id}`;
                document.getElementById('gameMeta').textContent = `PvP | Символ: ${currentGame.mySymbol}`;
                refreshGameStatus(data.state);
                document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));
                document.querySelectorAll('.panel').forEach(p => p.classList.remove('active'));
                document.querySelector('.tab-btn').classList.add('active');
                document.getElementById('panel-game').classList.add('active');
                startAutoRefresh(data.id);
            } catch (e) { alert(`Не удалось присоединиться: ${e.message}`); }
        }

        async function searchUser() {
            const query = document.getElementById('searchInput').value.trim();
            if (!query) { alert('Введите UUID'); return; }
            try {
                const data = await api(`/user/${query}`, 'GET');
                document.getElementById('foundLogin').textContent = data.login;
                document.getElementById('foundId').textContent = data.user_id;
                document.getElementById('searchResult').classList.remove('hidden');
            } catch (e) {
                document.getElementById('searchResult').classList.add('hidden');
                alert(`Пользователь не найден: ${e.message}`);
            }
        }

        window.addEventListener('DOMContentLoaded', () => {
            loadAuth();
            resetMatch();
            setInterval(() => {
                if (document.getElementById('panel-lobby').classList.contains('active')) {
                    refreshLobby();
                }
            }, 2000);
        });
        window.addEventListener('beforeunload', () => { stopAutoRefresh(); });
    </script>
