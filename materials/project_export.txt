import json
from typing import List, Optional
from sqlalchemy.orm import Mapped, mapped_column
from sqlalchemy import String, Integer
from datasource.db import Base

class GameSessionDB(Base):
    __tablename__: str = "game_sessions"
    session_id: Mapped[str] = mapped_column(String(36), primary_key=True)
    side_length: Mapped[int] = mapped_column(Integer, default=3)
    grid_json: Mapped[str] = mapped_column(String)
    state: Mapped[str] = mapped_column(String, default="waiting")
    player_x_id: Mapped[Optional[str]] = mapped_column(String(36), nullable=True)
    player_o_id: Mapped[Optional[str]] = mapped_column(String(36), nullable=True)
    mode: Mapped[str] = mapped_column(String, default="pve")

    def to_grid(self) -> List[List[int]]:
        return json.loads(self.grid_json)

    @classmethod
    def from_session(cls, session: "MatchSession", mode: str = "pve") -> "GameSessionDB":
        instance = cls()
        instance.session_id = str(session.session_id)
        instance.side_length = session.field.side_length
        instance.grid_json = json.dumps(session.field.grid)
        instance.state = session.state.value
        instance.player_x_id = session.player_x_id
        instance.player_o_id = session.player_o_id
        instance.mode = mode
        return instance

import json
from typing import Optional, Dict, Any, List
from sqlalchemy.orm import Session
from datasource.db import get_session
from datasource.model.game_db import GameSessionDB

class GameRepository:
    def __init__(self) -> None:
        pass

    def save_match(self, session_id: str, grid: list[list[int]], side_length: int, 
                   state: str = "waiting", player_x_id: Optional[str] = None,
                   player_o_id: Optional[str] = None, mode: str = "pve") -> None:
        db_session: Session = get_session()
        try:
            existing: Optional[GameSessionDB] = db_session.query(GameSessionDB).filter_by(session_id=session_id).first()
            if existing is not None:
                existing.side_length = side_length
                existing.grid_json = json.dumps(grid)
                existing.state = state
                existing.player_x_id = player_x_id
                existing.player_o_id = player_o_id
                existing.mode = mode
            else:
                new_record: GameSessionDB = GameSessionDB(
                    session_id=session_id,
                    side_length=side_length,
                    grid_json=json.dumps(grid),
                    state=state,
                    player_x_id=player_x_id,
                    player_o_id=player_o_id,
                    mode=mode
                )
                db_session.add(new_record)
            db_session.commit()
        except Exception:
            db_session.rollback()
            raise
        finally:
            db_session.close()

    def load_match(self, session_id: str) -> Optional[Dict[str, Any]]:
        db_session: Session = get_session()
        try:
            record: Optional[GameSessionDB] = db_session.query(GameSessionDB).filter_by(session_id=session_id).first()
            if record is not None:
                return {
                    "session_id": record.session_id,
                    "side_length": record.side_length,
                    "grid": record.to_grid(),
                    "state": record.state,
                    "player_x_id": record.player_x_id,
                    "player_o_id": record.player_o_id,
                    "mode": record.mode
                }
            else:
                return None
        finally:
            db_session.close()



    def find_waiting_games(self) -> List[Dict[str, Any]]:
        db_session: Session = get_session()
        try:
            records: List[GameSessionDB] = db_session.query(GameSessionDB).filter_by(
                state="waiting", 
                mode="pvp"
            ).all()
            return [
             {
                    "id": r.session_id,          
                    "session_id": r.session_id,   
                    "player_x_id": r.player_x_id,
                    "created_at": None
                }
                for r in records
            ]
        finally:
            db_session.close()

    def join_game(self, session_id: str, player_id: str) -> bool:
        db_session: Session = get_session()
        try:
            record: Optional[GameSessionDB] = db_session.query(GameSessionDB).filter_by(session_id=session_id).first()
            if record is not None and record.state == "waiting" and record.mode == "pvp":
                if record.player_x_id is None:
                    record.player_x_id = player_id
                elif record.player_o_id is None:
                    record.player_o_id = player_id
                else:
                    return False
                record.state = "player_x_turn"
                db_session.commit()
                return True
            else:
                return False
        except Exception:
            db_session.rollback()
            raise
        finally:
            db_session.close()

from typing import Any, Dict, Callable
from flask import Flask
from datasource.repository.game_repository import GameRepository

class DependencyBox:
    def __init__(self) -> None:
        self._singletons: Dict[str, Any] = {}
        self._factories: Dict[str, Callable[[], Any]] = {}
        self._setup()

    def _setup(self) -> None:
        def create_game_repo() -> GameRepository:
            return GameRepository()
        self.register_singleton("game_repo", create_game_repo)

        def create_user_repo() -> UserRepository:
            return UserRepository()
        self.register_singleton("user_repo", create_user_repo)

        def create_logic() -> LogicWithArchive:
            return LogicWithArchive(self.get("game_repo"))
        self.register("logic", create_logic)

        def create_user_svc() -> UserService:
            return UserService(self.get("user_repo"))
        self.register("user_svc", create_user_svc)

        def create_auth_svc() -> AuthService:
            return AuthService(self.get("user_svc"))
        self.register("auth_svc", create_auth_svc)

        def create_web_app() -> Flask:
            app: Flask = build_app(self.get("logic"), self.get("auth_svc"))
            app.config['game_repo'] = self.get("game_repo")
            app.config['user_repo'] = self.get("user_repo")
            app.config['logic_service'] = self.get("logic")
            app.register_blueprint(create_auth_routes())
            return app
        self.register("web_app", create_web_app)

    def register(self, name: str, factory: Callable[[], Any]) -> None:
        if name in self._singletons:
            del self._singletons[name]
        else:
            pass
        self._factories[name] = factory

    def register_singleton(self, name: str, factory: Callable[[], Any]) -> None:
        self._factories[name] = factory
        self._singletons[name] = factory()

    def get(self, name: str) -> Any:
        if name in self._singletons:
            return self._singletons[name]
        elif name in self._factories:
            return self._factories[name]()
        else:
            raise ValueError(f"Компонент '{name}' не найден в реестре")

import uuid
from enum import Enum
from typing import Optional
from domain.model.board import GameField

class GameState(Enum):
    WAITING = "waiting"
    PLAYER_X_TURN = "player_x_turn"
    PLAYER_O_TURN = "player_o_turn"
    DRAW = "draw"
    WIN_X = "win_x"
    WIN_O = "win_o"

class MatchSession:
    def __init__(
        self,
        current_field: GameField | None = None,
        player_x_id: Optional[str] = None,
        player_o_id: Optional[str] = None,
        state: GameState = GameState.WAITING
    ) -> None:
        self.session_id: uuid.UUID = uuid.uuid4()
        self.field: GameField = current_field or GameField()
        self.player_x_id: Optional[str] = player_x_id
        self.player_o_id: Optional[str] = player_o_id
        self.state: GameState = state

    def get_current_player_id(self) -> Optional[str]:
        if self.state == GameState.PLAYER_X_TURN:
            return self.player_x_id
        elif self.state == GameState.PLAYER_O_TURN:
            return self.player_o_id
        else:
            return None

    def get_opponent_symbol(self, player_id: str) -> Optional[int]:
        if player_id == self.player_x_id:
            return 2
        elif player_id == self.player_o_id:
            return 1
        else:
            return None

from typing import Tuple, List, Optional
from domain.service.rules_contract import GameRulesContract
from domain.model.game import MatchSession, GameState
from domain.model.board import GameField

class AiEngine(GameRulesContract):
    def __init__(self) -> None:
        pass

    def calculate_ai_turn(self, session: MatchSession) -> Tuple[int, int]:
        grid: List[List[int]] = [row.copy() for row in session.field.grid]
        top_score: float = float('-inf')
        best_cell: Tuple[int, int] = (-1, -1)

        for r in range(session.field.side_length):
            for c in range(session.field.side_length):
                if grid[r][c] == 0:
                    grid[r][c] = 2
                    score: int = self._run_minimax(grid, depth=0, is_max=False)
                    grid[r][c] = 0
                    if score > top_score:
                        top_score = score
                        best_cell = (r, c)
                    else:
                        continue
                else:
                    continue

        return best_cell

    def check_board_changes(self, current: MatchSession, previous: GameField) -> bool:
        if current.field.side_length != previous.side_length:
            return False
        else:
            change_count: int = 0
            for r in range(current.field.side_length):
                for c in range(current.field.side_length):
                    if current.field.grid[r][c] != previous.grid[r][c]:
                        change_count += 1
                    else:
                        continue
            if change_count == 1:
                return True
            else:
                return False

    def verify_end_state(self, session: MatchSession) -> bool:
        winner_code: int = self._find_winner(session.field.grid)
        if winner_code != 0:
            return True
        else:
            is_full: bool = all(cell != 0 for row in session.field.grid for cell in row)
            if is_full:
                return True
            else:
                return False

    def update_game_state(self, session: MatchSession) -> None:
        winner: int = self._find_winner(session.field.grid)
        if winner == 1:
            session.state = GameState.WIN_X
        elif winner == 2:
            session.state = GameState.WIN_O
        elif all(cell != 0 for row in session.field.grid for cell in row):
            session.state = GameState.DRAW
        elif hasattr(session, 'mode') and session.mode == "pve":
            if session.state == GameState.PLAYER_X_TURN:
                session.state = GameState.PLAYER_O_TURN
            else:
                session.state = GameState.PLAYER_X_TURN
        else:
            if session.state == GameState.PLAYER_X_TURN:
                session.state = GameState.PLAYER_O_TURN
            elif session.state == GameState.PLAYER_O_TURN:
                session.state = GameState.PLAYER_X_TURN
            else:
                session.state = GameState.PLAYER_X_TURN

    def _run_minimax(self, grid: List[List[int]], depth: int, is_max: bool) -> int:
        winner_code: int = self._find_winner(grid)
        if winner_code == 2:
            return 10 - depth
        elif winner_code == 1:
            return depth - 10
        elif all(cell != 0 for row in grid for cell in row):
            return 0
        else:
            pass

        if is_max:
            best: float = float('-inf')
            for r in range(3):
                for c in range(3):
                    if grid[r][c] == 0:
                        grid[r][c] = 2
                        score: int = self._run_minimax(grid, depth + 1, False)
                        grid[r][c] = 0
                        if score > best:
                            best = score
                        else:
                            continue
            return best
        else:
            best: float = float('inf')
            for r in range(3):
                for c in range(3):
                    if grid[r][c] == 0:
                        grid[r][c] = 1
                        score: int = self._run_minimax(grid, depth + 1, True)
                        grid[r][c] = 0
                        if score < best:
                            best = score
                        else:
                            continue
            return best

    @staticmethod
    def _find_winner(grid: List[List[int]]) -> int:
        lines: List[List[Tuple[int, int]]] = [
            [(0,0),(0,1),(0,2)], [(1,0),(1,1),(1,2)], [(2,0),(2,1),(2,2)],
            [(0,0),(1,0),(2,0)], [(0,1),(1,1),(2,1)], [(0,2),(1,2),(2,2)],
            [(0,0),(1,1),(2,2)], [(0,2),(1,1),(2,0)]
        ]
        for line in lines:
            cells: List[int] = [grid[r][c] for r, c in line]
            if cells[0] != 0 and cells[0] == cells[1] == cells[2]:
                return cells[0]
            else:
                continue
        return 0

import base64
from typing import Optional
from domain.service.user_service import UserService

class AuthService:
    def __init__(self, user_service: UserService) -> None:
        self._user_service: UserService = user_service

    def register(self, login: str, password: str) -> bool:
        if not login or not password:
            return False
        else:
            return self._user_service.create_user(login, password)

    def login(self, login: str, password: str) -> Optional[str]:
        if not login or not password:
            return None
        else:
            return self._user_service.verify_credentials(login, password)

    def authorize(self, auth_header: str) -> Optional[str]:
        if not auth_header.startswith("Basic "):
            return None
        else:
            try:
                encoded: str = auth_header.split(" ", 1)[1]
                decoded_bytes: bytes = base64.b64decode(encoded)
                decoded_str: str = decoded_bytes.decode("utf-8")
                parts: list[str] = decoded_str.split(":", 1)
                if len(parts) != 2:
                    return None
                else:
                    login: str = parts[0]
                    password: str = parts[1]
                    return self._user_service.verify_credentials(login, password)
            except Exception:
                return None

import uuid
from typing import Optional
from werkzeug.security import generate_password_hash, check_password_hash
from datasource.repository.user_repository import UserRepository
from datasource.model.user_db import UserDB

class UserService:
    def __init__(self, repository: UserRepository) -> None:
        self._repository: UserRepository = repository

    def create_user(self, login: str, raw_password: str) -> bool:
        existing: Optional[UserDB] = self._repository.find_by_login(login)
        if existing is not None:
            return False
        else:
            new_id: str = str(uuid.uuid4())
            hashed: str = generate_password_hash(raw_password)
            self._repository.save_user(new_id, login, hashed)
            return True

    def verify_credentials(self, login: str, raw_password: str) -> Optional[str]:
        record: Optional[UserDB] = self._repository.find_by_login(login)
        if record is not None:
            if check_password_hash(record.password_hash, raw_password):
                return record.user_id
            else:
                return None
        else:
            return None

from flask import Blueprint, request, jsonify, current_app
from web.model.auth_request import SignUpRequest
from domain.service.auth_service import AuthService

def create_auth_routes() -> Blueprint:
    bp = Blueprint('auth', __name__)

    @bp.route('/signup', methods=['POST'])
    def signup_endpoint() -> tuple:
        service: AuthService = current_app.config['auth_service']
        payload: dict | None = request.get_json()
        if payload is None:
            return jsonify({"error": "Пустое тело запроса"}), 400
        else:
            try:
                req: SignUpRequest = SignUpRequest(login=payload["login"], password=payload["password"])
            except KeyError:
                return jsonify({"error": "Отсутствуют поля login/password"}), 400

            result: bool = service.register(req.login, req.password)
            if result:
                return jsonify({"status": "registered"}), 201
            else:
                return jsonify({"error": "Логин уже занят"}), 409

    @bp.route('/login', methods=['POST'])
    def login_endpoint() -> tuple:
        service: AuthService = current_app.config['auth_service']
        payload: dict | None = request.get_json()
        
        if payload is None:
            return jsonify({"error": "Пустое тело запроса"}), 400
        
        login: str | None = payload.get("login")
        password: str | None = payload.get("password")
        
        if not login or not password:
            return jsonify({"error": "Отсутствуют поля login/password"}), 400

        return domain_session

from typing import Callable, Any
from functools import wraps
from flask import request, jsonify, current_app
from domain.service.auth_service import AuthService

class UserAuthenticator:
    def __init__(self, service: AuthService) -> None:
        self._service: AuthService = service

    def authenticate(self, func: Callable[..., Any]) -> Callable[..., Any]:
        @wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            header: str | None = request.headers.get('Authorization')
            if header is None:
                return jsonify({"error": "Требуется авторизация"}), 401
            else:
                user_id: str | None = self._service.authorize(header)
                if user_id is not None:
                    return func(*args, **kwargs)
                else:
                    return jsonify({"error": "Неверные учетные данные"}), 401
        return wrapper

        payload: Optional[Dict[str, Any]] = request.get_json()
        if payload is None or not isinstance(payload, dict):
            abort(400, description="Тело запроса должно быть объектом JSON")
        else:
            try:
                domain_session: MatchSession = ViewDomainTranslator.view_to_domain(SessionView(**payload))
            except (KeyError, ValueError, TypeError) as error:
                abort(400, description=f"Неверные данные игры: {str(error)}")

            prev_board: Optional[GameField] = service.load_previous_field(domain_session.session_id)
            if prev_board is not None:
                if not service.check_board_changes(domain_session, prev_board):
                    abort(400, description="Проверка поля не пройдена: изменены старые ходы")
                else:
                    pass
            else:
                pass

            if service.verify_end_state(domain_session):
                return respond(domain_session)
            else:
                row, col = service.calculate_ai_turn(domain_session)
                if row != -1 and domain_session.field.grid[row][col] == 0:

                    service.update_game_state(domain_session)

            if (!login || !password) { alert('Введите логин и пароль'); return; }
            try {
                await api('/signup', 'POST', { login, password }, false);
                alert('✓ Регистрация успешна! Теперь войдите.');
                log(`✓ Зарегистрирован: ${login}`, 'res');
            } catch (e) { alert(`Ошибка: ${e.message}`); }
        }

        async function doLogin() {
            const login = document.getElementById('loginInput').value.trim();
            const password = document.getElementById('passInput').value;
            if (!login || !password) { alert('Введите логин и пароль'); return; }
            try {
                const data = await api('/login', 'POST', { login, password }, false);
                saveAuth(login, password, data.user_id);
                alert('✓ Вход выполнен!');
            } catch (e) { alert(`Ошибка: ${e.message}`); }
        }

        // === Навигация ===
        function switchTab(tab) {
            document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));
            document.querySelectorAll('.panel').forEach(p => p.classList.remove('active'));
            event.target.classList.add('active');
            document.getElementById(`panel-${tab}`).classList.add('active');
            if (tab === 'lobby') refreshLobby();
            log(`→ Переключено на вкладку: ${tab}`, 'info');
        }

        // === Игра ===
        function resetMatch() {
            currentGame = null;
            field = Array(3).fill(null).map(() => Array(3).fill(0));
            active = true;
            drawBoard();
            setStatus('Ожидание создания игры...', 'waiting');
            document.getElementById('sid').textContent = '';
            document.getElementById('gameMeta').textContent = '';
            stopAutoRefresh();
        }

        function drawBoard() {
            const container = document.getElementById('board');
            container.innerHTML = '';
            for (let r = 0; r < 3; r++) {
                for (let c = 0; c < 3; c++) {
                    const btn = document.createElement('button');
                    btn.className = 'cell' + (field[r][c] === 1 ? ' x' : field[r][c] === 2 ? ' o' : '');
                    btn.textContent = field[r][c] === 1 ? 'X' : field[r][c] === 2 ? 'O' : '';
                    btn.disabled = !active || field[r][c] !== 0 || !currentGame;
                    btn.onclick = () => playerMove(r, c);
                    container.appendChild(btn);
                }
            }
        }

        function setStatus(text, cls = '') {
            const el = document.getElementById('gameStatus');
            el.textContent = text;
            el.className = `status ${cls}`;
        }

        async function createGame(mode) {
            if (!currentUser) { alert('Сначала войдите в аккаунт'); return; }
            try {
                const data = await api('/game', 'POST', { mode });
                currentGame = { id: data.id, mode: data.mode, state: data.state, mySymbol: mode === 'pvp' ? 'X' : 'X' };
                field = data.board.grid.map(row => [...row]);
                active = data.state === 'player_x_turn' || data.state === 'waiting';
                drawBoard();
                document.getElementById('sid').textContent = `ID игры: ${data.id}`;
                document.getElementById('gameMeta').textContent = `Режим: ${mode.toUpperCase()} | Символ: ${currentGame.mySymbol}`;
                setStatus(mode === 'pvp' ? 'Ожидание соперника...' : 'Твой ход (X)', mode === 'pvp' ? 'waiting' : 'turn');
                if (mode === 'pve') startAutoRefresh(data.id);
                log(`✓ Создана игра: ${data.id} (${mode})`, 'res');
            } catch (e) { alert(`Ошибка: ${e.message}`); }
        }

        async function playerMove(r, c) {
            if (!active || !currentGame || field[r][c] !== 0) return;
            field[r][c] = 1;
            drawBoard();
            setStatus('Отправка хода...', 'waiting');
            try {
                const data = await api(`/game/${currentGame.id}`, 'POST', {
                    id: currentGame.id,
                    board: { grid: field, size: 3 }
                });
                field = data.board.grid.map(row => [...row]);
                drawBoard();
                checkWin(data.state);
                if (currentGame.mode === 'pve' && data.state !== 'win_x' && data.state !== 'draw') {
                    setStatus('Ход компьютера...', 'waiting');
                }
            } catch (e) {
                field[r][c] = 0;
                drawBoard();
                setStatus(`Ошибка: ${e.message}`, '');
            }
        }

        function checkWin(state) {
            const lines = [
                [[0,0],[0,1],[0,2]],[[1,0],[1,1],[1,2]],[[2,0],[2,1],[2,2]],
                [[0,0],[1,0],[2,0]],[[0,1],[1,1],[2,1]],[[0,2],[1,2],[2,2]],
                [[0,0],[1,1],[2,2]],[[0,2],[1,1],[2,0]]
            ];
            for (const [[r1,c1],[r2,c2],[r3,c3]] of lines) {
                if (field[r1][c1] && field[r1][c1] === field[r2][c2] && field[r2][c2] === field[r3][c3]) {
                    const w = field[r1][c1];
                    if (w === 1) { setStatus('🎉 Ты победил!', 'win'); active = false; }
                    else { setStatus('💻 Компьютер выиграл', 'lose'); active = false; }
                    stopAutoRefresh();
                    return;
                }
            }
            if (state === 'draw') { setStatus('🤝 Ничья!', 'draw'); active = false; stopAutoRefresh(); return; }
            if (field.flat().every(v => v !== 0)) { setStatus('🤝 Ничья!', 'draw'); active = false; stopAutoRefresh(); return; }
            if (currentGame.mode === 'pve') {
                setStatus('Твой ход (X)', 'turn');
            } else {
                setStatus(state === 'player_o_turn' ? 'Ход соперника (O)' : 'Твой ход (X)', 'turn');
            }
        }

        async function fetchGameState(gameId) {
            if (!currentGame || currentGame.id !== gameId) return;
            try {
                const data = await api(`/game/${gameId}`, 'GET');
                field = data.board.grid.map(row => [...row]);
                currentGame.state = data.state;
                drawBoard();
                checkWin(data.state);
            } catch (e) { /* игнорируем ошибки автообновления */ }
        }

        function startAutoRefresh(gameId) {
            stopAutoRefresh();
            autoRefresh = setInterval(() => fetchGameState(gameId), 2000);
            log('↻ Автообновление игры включено (2 сек)', 'info');
        }

        function stopAutoRefresh() {
            if (autoRefresh) { clearInterval(autoRefresh); autoRefresh = null; log('⏹ Автообновление остановлено', 'info'); }
        }

        // === Лобби (обновлённое) ===
        async function refreshLobby() {
            if (!currentUser) { 
                document.getElementById('lobbyList').innerHTML = '<div class="game-item"><small>🔐 Войдите для доступа к лобби</small></div>'; 
                return; 
            }
            try {
                const data = await api('/games/available', 'GET');
                // ✅ Исправлено: используем поле "id" из ответа бэкенда
                lobbyGames = (data.games || []).map(g => ({
                    id: g.id || g.session_id,  // Поддержка обоих форматов
                    session_id: g.session_id,
                    player_x_id: g.player_x_id,
                    created_at: g.created_at
                }));
                renderLobbyList(lobbyGames);
                updateLobbyStats();
                log(`✓ Загружено игр: ${lobbyGames.length}`, 'res');
            } catch (e) {
                document.getElementById('lobbyList').innerHTML = `<div class="game-item"><small class="err">❌ ${e.message}</small></div>`;
            }
        }

        function renderLobbyList(games) {
            const list = document.getElementById('lobbyList');
            const filtered = games.filter(g => 
                !lobbyFilter || g.id.toLowerCase().includes(lobbyFilter.toLowerCase())
            );
            
            if (filtered.length === 0) {
                list.innerHTML = '<div class="game-item"><small>📭 Нет доступных игр. Создайте свою!</small></div>';
                return;
            }
            
            list.innerHTML = filtered.map(g => `
                <div class="game-item">
                    <div>
                        <strong>ID:</strong> <code>${g.id.slice(0, 8)}...</code>
                        ${g.player_x_id ? `<br><small>Создатель: ${g.player_x_id.slice(0, 8)}...</small>` : ''}
                    </div>
                    <button class="btn secondary" onclick="joinGame('${g.id}')">🎮 Присоединиться</button>
                </div>
            `).join('');
        }

        function filterLobby() {
            lobbyFilter = document.getElementById('lobbySearch').value.trim();
            renderLobbyList(lobbyGames);
            log(`🔍 Фильтр лобби: "${lobbyFilter}"`, 'info');
        }

        function clearLobbyFilter() {
            lobbyFilter = '';
            document.getElementById('lobbySearch').value = '';
            renderLobbyList(lobbyGames);
            log('🔄 Фильтр сброшен', 'info');
        }

        function updateLobbyStats() {
            const stats = document.getElementById('lobbyStats');
            const shown = lobbyGames.filter(g => 
                !lobbyFilter || g.id.toLowerCase().includes(lobbyFilter.toLowerCase())
            ).length;
            stats.textContent = `Показано: ${shown} из ${lobbyGames.length} игр`;
        }

        // === Поиск активных игр ===
        async function searchActiveGames() {
            if (!currentUser) { alert('Сначала войдите'); return; }
            
            const mode = document.getElementById('searchMode').value;
            const state = document.getElementById('searchState').value;
            const gameId = document.getElementById('searchGameId').value.trim();
            
            try {
                const data = await api('/games/available', 'GET');
                let results = (data.games || []).map(g => ({
                    id: g.id || g.session_id,
                    session_id: g.session_id,
                    player_x_id: g.player_x_id,
                    mode: 'pvp',  // /games/available возвращает только PvP
                    state: 'waiting'
                }));
                
                // Применяем фильтры
                if (mode !== 'all' && mode !== 'pvp') {
                    results = [];  // PvE-игры не в лобби
                }
                if (state !== 'all' && state !== 'waiting') {
                    results = [];  // Только waiting в лобби
                }
                if (gameId) {
                    results = results.filter(g => g.id.toLowerCase().includes(gameId.toLowerCase()));
                }
                
                renderSearchResults(results);
                log(`🔍 Найдено игр: ${results.length}`, 'res');
            } catch (e) {
                document.getElementById('searchResults').innerHTML = `<div class="game-item"><small class="err">❌ ${e.message}</small></div>`;
            }
        }

        function renderSearchResults(games) {
            const container = document.getElementById('searchResults');
            if (games.length === 0) {
                container.innerHTML = '<div class="game-item"><small>📭 Игры не найдены</small></div>';
                return;
            }
            container.innerHTML = games.map(g => `
                <div class="game-item">
                    <div>
                        <strong>ID:</strong> <code>${g.id}</code>
                        <br><small>Режим: ${g.mode.toUpperCase()} | Статус: ${g.state}</small>
                        ${g.player_x_id ? `<br><small>Создатель: ${g.player_x_id.slice(0, 12)}...</small>` : ''}
                    </div>
                    <button class="btn" onclick="joinGame('${g.id}')">🎮 Присоединиться</button>
                </div>
            `).join('');
        }

        async function joinGame(gameId) {
            if (!currentUser) { alert('Сначала войдите'); return; }
            try {
                const data = await api(`/game/${gameId}/join`, 'POST', null);
                currentGame = { 
                    id: data.id, 
                    mode: 'pvp', 
                    state: data.state, 
                    mySymbol: data.player_x_id === currentUser.userId ? 'X' : 'O' 
                };
                field = data.board.grid.map(row => [...row]);
                active = data.state.includes('turn');
                drawBoard();
                document.getElementById('sid').textContent = `ID игры: ${data.id}`;
                document.getElementById('gameMeta').textContent = `PvP | Символ: ${currentGame.mySymbol}`;
                setStatus(data.state === 'player_x_turn' || data.state === 'player_o_turn' ? 'Твой ход!' : 'Ожидание хода соперника...', 'waiting');
                switchTab('game');
                startAutoRefresh(data.id);
                log(`✓ Присоединён к игре: ${gameId}`, 'res');
            } catch (e) { alert(`Не удалось присоединиться: ${e.message}`); }
        }

        // === Поиск пользователя ===
        async function searchUser() {
            const query = document.getElementById('searchInput').value.trim();
            if (!query) { alert('Введите UUID или логин'); return; }
            try {
                const data = await api(`/user/${query}`, 'GET');
                document.getElementById('foundLogin').textContent = data.login;
                document.getElementById('foundId').textContent = data.user_id;
                document.getElementById('searchResult').classList.remove('hidden');
                log(`✓ Найден пользователь: ${data.login}`, 'res');
            } catch (e) {
                document.getElementById('searchResult').classList.add('hidden');
                alert(`Пользователь не найден: ${e.message}`);
            }
        }
